1. Maximum and Minimum Among Three Numbers

#include <stdio.h>

int main()
{
    int a, b, c, max, min;

    printf("Enter three numbers: ");
    scanf("%d%d%d", &a, &b, &c);

    max = min = a;

    if (b > max)
        max = b;

    if (c > max)
        max = c;

    if (b < min)
        min = b;

    if (c < min)
        min = c;

    printf("Maximum = %d\n", max);
    printf("Minimum = %d\n", min);

    return 0;
}

2. Simple Interest and Compound Interest

#include <stdio.h>
#include <math.h>

int main()
{
    float p, r, t, si, ci, amount;

    printf("Enter principal, rate and time: ");
    scanf("%f%f%f", &p, &r, &t);

    si = (p * r * t) / 100;
    amount = p * pow(1 + r / 100, t);
    ci = amount - p;

    printf("Simple Interest = %.2f\n", si);
    printf("Compound Interest = %.2f\n", ci);

    return 0;
}

3. Expression Evaluation

14. Calculator Using Switch Statement

#include <stdio.h>

int main()
{
    int a, b;
    char op;

    printf("Enter expression: ");
    scanf("%d %c %d", &a, &op, &b);

    switch (op)
    {
        case '+':
            printf("Result = %d\n", a + b);
            break;

        case '-':
            printf("Result = %d\n", a - b);
            break;

        case '*':
            printf("Result = %d\n", a * b);
            break;

        case '/':
            if (b != 0)
                printf("Result = %.2f\n", (float)a / b);
            else
                printf("Division by zero is not possible\n");
            break;

        case '%':
            if (b != 0)
                printf("Result = %d\n", a % b);
            else
                printf("Modulo by zero is not possible\n");
            break;

        default:
            printf("Invalid operator\n");
    }

    return 0;
}

15. Prime Number

#include <stdio.h>

int main()
{
    int n, i, isPrime = 1;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n <= 1)
        isPrime = 0;
    else
    {
        for (i = 2; i <= n / i; i++)
        {
            if (n % i == 0)
            {
                isPrime = 0;
                break;
            }
        }
    }

    if (isPrime == 1)
        printf("Prime Number\n");
    else
        printf("Not a Prime Number\n");

    return 0;
}

16. Sum of Digits and Number Palindrome

#include <stdio.h>

int main()
{
    int n, temp, digit;
    int sum = 0, reverse = 0;

    printf("Enter a positive number: ");
    scanf("%d", &n);

    temp = n;

    while (temp > 0)
    {
        digit = temp % 10;
        sum = sum + digit;
        reverse = reverse * 10 + digit;
        temp = temp / 10;
    }

    printf("Sum of digits = %d\n", sum);

    if (n == reverse)
        printf("Palindrome Number\n");
    else
        printf("Not a Palindrome Number\n");

    return 0;
}

18. Minimum, Maximum and Average in an Array

#include <stdio.h>

int main()
{
    int a[100], n, i;
    int min, max, sum = 0;
    float average;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements: ");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
        sum = sum + a[i];
    }

    min = max = a[0];

    for (i = 1; i < n; i++)
    {
        if (a[i] < min)
            min = a[i];

        if (a[i] > max)
            max = a[i];
    }

    average = (float)sum / n;

    printf("Minimum = %d\n", min);
    printf("Maximum = %d\n", max);
    printf("Average = %.2f\n", average);

    return 0;
}

19. Addition and Multiplication of Two Matrices Using Functions

#include <stdio.h>

void addition(int a[2][2], int b[2][2])
{
    int i, j;

    printf("Matrix Addition:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
            printf("%d ", a[i][j] + b[i][j]);

        printf("\n");
    }
}

void multiplication(int a[2][2], int b[2][2])
{
    int i, j, k, sum;

    printf("Matrix Multiplication:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 2; j++)
        {
            sum = 0;

            for (k = 0; k < 2; k++)
                sum = sum + a[i][k] * b[k][j];

            printf("%d ", sum);
        }

        printf("\n");
    }
}

int main()
{
    int a[2][2], b[2][2];
    int i, j;

    printf("Enter first matrix elements:\n");

    for (i = 0; i < 2; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);

    printf("Enter second matrix elements:\n");

    for (i = 0; i < 2; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &b[i][j]);

    addition(a, b);
    multiplication(a, b);

    return 0;
}

5. Strings

22 (I). Insert a Substring into a Given String at a Position

#include <stdio.h>
#include <string.h>

void insert(char str[], char sub[], int pos)
{
    char result[300];
    int i, j, k = 0;

    for (i = 0; i < pos; i++)
    {
        result[k] = str[i];
        k++;
    }

    for (j = 0; sub[j] != '\0'; j++)
    {
        result[k] = sub[j];
        k++;
    }

    for (i = pos; str[i] != '\0'; i++)
    {
        result[k] = str[i];
        k++;
    }

    result[k] = '\0';
    strcpy(str, result);
}

int main()
{
    char str[300], sub[150];
    int pos;

    printf("Enter main string: ");
    fgets(str, sizeof(str), stdin);
    str[strcspn(str, "\n")] = '\0';

    printf("Enter substring: ");
    fgets(sub, sizeof(sub), stdin);
    sub[strcspn(sub, "\n")] = '\0';

    printf("Enter position: ");
    scanf("%d", &pos);

    insert(str, sub, pos);

    printf("Result = %s\n", str);

    return 0;
}

25. Count Lines, Words and Characters Without ctype.h

#include <stdio.h>

int main()
{
    char ch;

    int lines = 0, words = 0, characters = 0;
    int inWord = 0;

    printf("Enter text (Ctrl+Z then Enter to stop):\n");

    while ((ch = getchar()) != EOF)
    {
        characters++;

        if (ch == '\n')
            lines++;

        if (ch == ' ' || ch == '\n' || ch == '\t' || ch == '\r')
        {
            inWord = 0;
        }
        else if (inWord == 0)
        {
            words++;
            inWord = 1;
        }

    printf("Words = %d\n", words);
    printf("Characters = %d\n", characters);

    return 0;
}

In Windows, press Ctrl + Z and then Enter to stop entering text. This program counts newline-terminated lines.

6. Sorting and Searching

26. Linear Search Using Non-Recursive Function

#include <stdio.h>

int linearSearch(int a[], int n, int key)
{
    int i;

    for (i = 0; i < n; i++)
    {
        if (a[i] == key)
            return i;
    }

    return -1;
}

int main()
{
    int a[100], n, i, key, pos;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter key element: ");
    scanf("%d", &key);

    pos = linearSearch(a, n, key);

    if (pos == -1)
        printf("Element not found\n");
    else
        printf("Element found at position %d\n", pos + 1);

    return 0;
}

27. Binary Search Using Non-Recursive Function

#include <stdio.h>

int binarySearch(int a[], int n, int key)
{
    int low = 0, high = n - 1;
    int mid;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (a[mid] == key)
            return mid;
        else if (key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    return -1;
}

int main()
{
    int a[100], n, i, key, pos;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter sorted array elements:\n");

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter key element: ");
    scanf("%d", &key);

    pos = binarySearch(a, n, key);

    if (pos == -1)
        printf("Element not found\n");
    else
        printf("Element found at position %d\n", pos + 1);

    return 0;
}

Binary search requires the array to be sorted in ascending order.

28. Bubble Sort in Ascending Order

#include <stdio.h>

int main()
{
    int a[100], n, i, j, temp;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Bubble Sort in Ascending Order
    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - 1 - i; j++)
        {
            if (a[j] > a[j + 1])
            {
                temp = a[j];

                a[j] = a[j + 1];

                a[j + 1] = temp;
            }
        }
    }

    printf("Ascending order:\n");

    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

29. Selection Sort in Descending Order

#include <stdio.h>

int main()
{
    int a[100], n, i, j;
    int maxIndex, temp;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    // Selection Sort in Descending Order
    for (i = 0; i < n - 1; i++)
    {
        maxIndex = i;

        for (j = i + 1; j < n; j++)
        {
            if (a[j] > a[maxIndex])
            {
                maxIndex = j;
            }
        }

        temp = a[i];

        a[i] = a[maxIndex];

        a[maxIndex] = temp;
    }

    printf("Descending order:\n");

    for (i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
